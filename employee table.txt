  employee_id INT PRIMARY KEY,
  first_name VARCHAR(50),
  last_name VARCHAR(50),
  department VARCHAR(50),
  salary INT,
  hire_date DATE,
  position VARCHAR(50)
);

Task	SQL Command
1. Select all data	SELECT * FROM Employee;
2. Employees in IT dept	SELECT * FROM Employee WHERE department = 'IT';
3. Count employees in each department	SELECT department, COUNT(*) FROM Employee GROUP BY department;
4. Avg salary per department	SELECT department, AVG(salary) FROM Employee GROUP BY department;
5. Employees hired after 1 Feb 2021	SELECT * FROM Employee WHERE hire_date > '2021-02-01';
6. Increase IT salary by ₹5000	UPDATE Employee SET salary = salary + 5000 WHERE department = 'IT';
7. Highest salary in each department	SELECT department, MAX(salary) FROM Employee GROUP BY department;
8. Departments with more than 1 employee	SELECT department, COUNT(*) FROM Employee GROUP BY department HAVING COUNT(*) > 1;
9. Employee with highest/lowest salary	SELECT * FROM Employee ORDER BY salary DESC LIMIT 1;
